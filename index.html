const loader = new THREE.GLTFLoader(); loader.load( '3d puppy.glb', function (gltf) { const model = gltf.scene; // --- 1. 축 방향 강제 교정 (누워있는 강아지 세우기) --- // X축을 기준으로 -90도(Math.PI / 2) 회전시킵니다. // 만약 이대로 했는데도 뒤집혀 있다면 -Math.PI / 2 대신 Math.PI / 2 로 바꿔보세요. // 옆으로 누워있다면 model.rotation.z = -Math.PI / 2; 를 추가해보세요. model.rotation.x = -Math.PI / 2; // --- 2. 모델 자동 중앙 정렬 및 크기 맞춤 --- const box = new THREE.Box3().setFromObject(model); const center = box.getCenter(new THREE.Vector3()); const size = box.getSize(new THREE.Vector3()); const maxDim = Math.max(size.x, size.y, size.z); const targetSize = 3.5; const scale = targetSize / maxDim; model.scale.setScalar(scale); model.position.sub(center.multiplyScalar(scale)); model.position.y -= 0.5; dogGroup.add(model); document.getElementById('loading-indicator').style.display = 'none'; document.getElementById('startBtn').classList.remove('hidden'); }, undefined, function (error) { console.error('모델 로드 실패:', error); document.getElementById('loading-indicator').innerText = '모델을 불러오지 못했습니다. 파일명을 확인해주세요.'; } );
